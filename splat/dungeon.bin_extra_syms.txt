/* Hand-maintained extra linker symbols for dungeon.bin, passed to ld alongside
   splat's two generated `*_auto.txt` scripts (see the Makefile's dungeon
   rule).  Nothing here is generated; splat never writes this file, so an
   entry added here survives `make splat`.

   D_800DF37A is a 2-byte datum that splat did not name.  Its neighbours
   D_800DF374, D_800DF378 and D_800DF37C are all in
   dungeon.bin_undefined_syms_auto.txt; 0x800DF37A fell through because
   nothing in the already-split code referenced it directly.

   The witness is func_800BA810, which reaches 0x800DF37A THREE times, each
   through its own `lui %hi` / `%lo` pair.  cse shares one `%hi` across a
   symbol and a constant offset, so every spelling that reaches the address as
   D_800DF378 + 2 -- a two-u16 struct, a `u16 x[]` array, and
   `*(u16 *)((u8 *)&D_800DF378 + 2)` were all measured -- collapses the three
   pairs into one and cannot get below 12 differing.  Only an independent
   symbol reproduces three independent pairs.  The raw constant address
   `*(u16 *)0x800DF37A` fixes the loads but turns the store into the
   assembler's absolute macro (`lui $at`).  See docs/matching-levers.md,
   "A SYMBOL PLUS A CONSTANT OFFSET IS ONE %hi TO cse; TWO SYMBOLS ARE TWO". */

D_800DF37A = 0x800DF37A;

/* THE TWO RODATA ISLANDS THE CARVE NAMED AS FUNCTIONS, 2026-11-23.

   splat/dungeon.bin.yaml records that b31f and lo20 carry rodata ISLANDS
   between their functions and that the function after each island loses its
   own symbol.  This is the other half of that cost, and it is a LATENT
   failure: the committed tree does not relink.

   At each of these two addresses spimdisasm emits `dlabel func_<tag>_<vram>`
   -- the carve's function name, on a DATA label -- while the code that
   reaches the address gets splat's data spelling, `D_<tag>_<vram>`.  The two
   names never meet and the dungeon link dies on `undefined reference to
   D_b31f_80174674'.  Nothing catches it until something re-runs splat, which
   is why it survived the carve's own byte-identical build: the listings that
   were on disk then had been written by an earlier run.

   A grep for `^dlabel func_' over the carved listing directories finds
   exactly these two, so this list is complete as carved.  (Do not write that
   glob with its trailing slash-star here: the star-slash ENDS THIS COMMENT,
   ld parses the rest of the prose as script and says only `syntax error'.)  The real fix is a rodata
   subsegment for each of the two objects, which the yaml already says they
   are worth; until somebody reads them, an absolute symbol costs nothing --
   the bytes themselves are emitted by the INCLUDE_ASM at that VMA, exactly as
   for D_800DF37A above. */

D_b31f_80174674 = 0x80174674;
D_lo20_8001E814 = 0x8001E814;
